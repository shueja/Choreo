cmake_minimum_required(VERSION 3.21)
project(choreo_document_consumer LANGUAGES CXX)

include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared libraries" FORCE)
set(WPILIB_WITH_CSCORE OFF CACHE INTERNAL "Build CSCore")
set(WPILIB_WITH_GUI OFF CACHE INTERNAL "Build GUI")
set(WPILIB_WITH_NTCORE ON CACHE INTERNAL "Build NTCore")
set(WPILIB_WITH_SIMULATION_MODULES OFF CACHE INTERNAL "Build simulation modules")
set(WPILIB_WITH_TESTS OFF CACHE INTERNAL "Build WPILib tests")
set(WPILIB_WITH_WPIMATH ON CACHE INTERNAL "Build WPIMath")
set(WPILIB_WITH_WPILIB OFF CACHE INTERNAL "Build WPILib")
set(WPILIB_NO_WERROR ON CACHE INTERNAL "Disable WPILib warnings as errors")
set(WPILIB_USE_SYSTEM_EIGEN ON CACHE BOOL "Use fetched Eigen")
set(EIGEN_BUILD_CMAKE_PACKAGE TRUE)
FetchContent_Declare(Eigen3 GIT_REPOSITORY https://github.com/eigen-mirror/eigen.git
    GIT_TAG b7f6aed1b9974fd56cce651b4ff059c3bfc67782 SYSTEM OVERRIDE_FIND_PACKAGE)
FetchContent_MakeAvailable(Eigen3)
set(WPILIB_USE_SYSTEM_SLEIPNIR OFF CACHE INTERNAL "Use bundled Sleipnir")
FetchContent_Declare(wpilib GIT_REPOSITORY https://github.com/wpilibsuite/allwpilib.git
    GIT_TAG f06afad7b7bd3fa2ef94a6c0c7d0174fe207750a EXCLUDE_FROM_ALL SYSTEM UPDATE_DISCONNECTED TRUE)
FetchContent_MakeAvailable(wpilib)
if(TARGET wpimath AND NOT TARGET wpimath::wpimath)
    add_library(wpimath::wpimath ALIAS wpimath)
endif()
if(TARGET wpiutil AND NOT TARGET wpiutil::wpiutil)
    add_library(wpiutil::wpiutil ALIAS wpiutil)
endif()

option(CHOREO_CONSUMER_APP "Test the app package" OFF)
option(CHOREO_CONSUMER_BUILD_APP_SOURCE "Build the app project against installed dependencies" OFF)
find_package(choreo-document-core CONFIG REQUIRED)
get_target_property(core_dependencies choreo::document_core INTERFACE_LINK_LIBRARIES)
if(core_dependencies MATCHES "immer|wpilib::expression|units::units")
    message(FATAL_ERROR "Robot-safe document target exposes app-only dependencies")
endif()
add_executable(choreo_document_consumer main.cpp)
target_link_libraries(choreo_document_consumer PRIVATE choreo::document_core)
if(CHOREO_CONSUMER_APP)
    if(CHOREO_CONSUMER_BUILD_APP_SOURCE)
        set(CHOREO_APP_USE_INSTALLED_DEPENDENCIES ON CACHE BOOL "Use installed dependencies" FORCE)
        add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../cpp/choreo_app" choreo-app)
    else()
        find_package(choreo-document-app CONFIG REQUIRED)
    endif()
    add_executable(choreo_document_app_consumer main.cpp)
    target_compile_definitions(choreo_document_app_consumer PRIVATE CHOREO_CONSUMER_APP)
    target_link_libraries(choreo_document_app_consumer PRIVATE choreo::document_app)
endif()
enable_testing()
add_test(NAME consumer.robot COMMAND choreo_document_consumer)
if(CHOREO_CONSUMER_APP)
    add_test(NAME consumer.app COMMAND choreo_document_app_consumer)
endif()
