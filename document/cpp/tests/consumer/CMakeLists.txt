cmake_minimum_required(VERSION 3.21)
project(choreo_document_consumer LANGUAGES CXX)

include(FetchContent)
include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/FetchWPILib.cmake")
fetch_wpilib()

option(CHOREO_CONSUMER_APP "Test the app package" OFF)
option(CHOREO_CONSUMER_BUILD_APP_SOURCE "Build the app project against installed dependencies" OFF)
find_package(choreo-document-core CONFIG REQUIRED)
get_target_property(core_dependencies choreo::document_core INTERFACE_LINK_LIBRARIES)
if(core_dependencies MATCHES "immer|wpilib::expression|units::units")
    message(FATAL_ERROR "Robot-safe document target exposes app-only dependencies")
endif()
add_executable(choreo_document_consumer main.cpp)
target_link_libraries(choreo_document_consumer PRIVATE choreo::document_core)
if(CHOREO_CONSUMER_APP)
    if(CHOREO_CONSUMER_BUILD_APP_SOURCE)
        set(CHOREO_APP_USE_INSTALLED_DEPENDENCIES ON CACHE BOOL "Use installed dependencies" FORCE)
        add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../cpp/choreo_app" choreo-app)
    else()
        find_package(choreo-document-app CONFIG REQUIRED)
    endif()
    add_executable(choreo_document_app_consumer main.cpp)
    target_compile_definitions(choreo_document_app_consumer PRIVATE CHOREO_CONSUMER_APP)
    target_link_libraries(choreo_document_app_consumer PRIVATE choreo::document_app)
endif()
enable_testing()
add_test(NAME consumer.robot COMMAND choreo_document_consumer)
if(CHOREO_CONSUMER_APP)
    add_test(NAME consumer.app COMMAND choreo_document_app_consumer)
endif()
